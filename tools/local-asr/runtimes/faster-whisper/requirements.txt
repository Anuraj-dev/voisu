#   uv pip install --python $VENV/bin/python faster-whisper==1.2.1 ctranslate2==4.8.2 \

# which the adapter preloads from this venv.

# 5560830fe8fa09b8cdf1b5926d0cc39a2586eadaad685d809e156432c4ebca80); two
# conversions produced identical bytes. Converter environment:
# ctranslate2==4.8.2, transformers==4.56.1, torch==2.9.1+cu128 (CPU only).

#     --copy_files tokenizer.json preprocessor_config.json \
#     --quantization float16 --low_cpu_mem_usage
